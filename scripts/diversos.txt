// Estrutura para armazenar o código de cada questão para um aluno. 
// topico: contem dados de um topico do classroom 
// sourcecode: é o codigo fonte da submissao feirta pelo aluno e obtido do drive do professor
// exercicio é o id do classroom de uma das submissoes do aluno dentro do topico (classwork) classroom
// subQuestions: as 3 perguntas/sub-questoes que serao criadas pela api openai
interface QuestionData {
  topico: Coursework;
  sourceCode: string;
  exercicio: submissionId;
  subQuestions: string[];
}

// dados de uma atividade prática  obtidos do classroom
interface Coursework {
  id: string;
  name: string;
}

// dados de um aluno selecionado no frontend
// id do aluno obtido do classroom
// question data, array, cada elemento corresponde a uma questao obtida do numero de questoes 
interface Student {
  id: string;
  name: string;
  question_data: QuestionData[]
}

// id e name obtidos de dados do curso do classroom
interface Course {
  name: string (obtido da api classroom)
  id: obtido do classroom
}

interface Evaluation {
  id: (unico, gerado pelo frontend)
  course: Course;
  title: string;
  date: string;
  numQuestions: number;
  selectedCourseworks: string[];
  selectedStudents: Student[];
}
